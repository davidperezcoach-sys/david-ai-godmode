function detectarSolicitudImagen(mensaje: string): boolean {
  const triggers = [
    "genera una imagen","genera imagen","crea una imagen","crea imagen",
    "hazme una imagen","haz una imagen","quiero una imagen","ponme una imagen",
    "dibuja","visualiza","muéstrame","imagen de","foto de","ilustra",
    "genera un dibujo","crea un dibujo","generate image","create image"
  ];
  return triggers.some(t => mensaje.toLowerCase().includes(t));
}

function ariaDeberiaGenerarImagen(mensaje: string, respuesta: string, pagina: string): boolean {
  const paginasVisuales = ["ejercicio","visualizacion","meditacion","respiracion","herramientas"];
  const esEjercicioVisual = paginasVisuales.some(p => pagina?.toLowerCase().includes(p));
  const respuestaVisual = /(visualiza|imagina|cierra los ojos|respira|siente|observa|contempla)/i.test(respuesta);
  const completoEjercicio = /(completaste|terminaste|lo lograste|acabas de|bien hecho|lo has hecho)/i.test(respuesta);
  return esEjercicioVisual && (respuestaVisual || completoEjercicio);
}

async function generarImagenDalle(prompt: string, apiKey: string): Promise<string | null> {
  try {
    const res = await fetch(DALLE_URL, {
      method: "POST",
      headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json" },
      body: JSON.stringify({ model: "dall-e-3", prompt, n: 1, size: "1024x1024", quality: "standard", response_format: "url" })
    });
    const data = await res.json();
    return data?.data?.[0]?.url || null;
  } catch(e) {
    console.error("[ARIA-DALLE] Error:", e);
    return null;
  }
}

interface EstadoCognitivo {
  nivelEmocional: number;        // 0-10
  patronDetectado: string;       // nombre del patrón
  urgencia: "baja" | "media" | "alta" | "critica";
  señales: string[];
}

function analizarEstadoCognitivo(mensaje: string): EstadoCognitivo {
  const texto = mensaje.toLowerCase();
  const señales: string[] = [];
  let nivel = 5;

  // Señales de intensidad emocional alta
  if (/!!!|!!|\?\?\?/.test(mensaje)) { señales.push("intensidad_alta"); nivel += 2; }
  if (/todo mayúsculas/i.test(mensaje) || mensaje === mensaje.toUpperCase() && mensaje.length > 10) { señales.push("tension_maxima"); nivel += 3; }
  if (/no puedo|no sé|me rindo|rendirse|ya no|nunca|siempre fallo/i.test(texto)) { señales.push("desesperanza"); nivel += 2; }
  if (/sola|solo|nadie|abandonad/i.test(texto)) { señales.push("aislamiento"); nivel += 2; }
  if (/miedo|terror|pánico|angustia|ansiedad|panico/i.test(texto)) { señales.push("miedo_activo"); nivel += 1; }
  if (/bien|genial|feliz|alegr|contento|emocionad/i.test(texto)) { señales.push("estado_positivo"); nivel -= 2; }
  if (/gracias|logré|conseguí|pude|lo hice/i.test(texto)) { señales.push("logro_reciente"); nivel -= 1; }

  nivel = Math.max(0, Math.min(10, nivel));

  let patronDetectado = "neutro";
  if (señales.includes("desesperanza")) patronDetectado = "creencia_limitante_activa";
  else if (señales.includes("aislamiento")) patronDetectado = "herida_abandono";
  else if (señales.includes("miedo_activo")) patronDetectado = "respuesta_supervivencia";
  else if (señales.includes("logro_reciente")) patronDetectado = "momentum_positivo";
  else if (señales.includes("estado_positivo")) patronDetectado = "apertura_receptiva";

  const urgencia: EstadoCognitivo["urgencia"] =
    nivel >= 9 ? "critica" :
    nivel >= 7 ? "alta" :
    nivel >= 4 ? "media" : "baja";

  return { nivelEmocional: nivel, patronDetectado, urgencia, señales };
}

function detectarComandoExterno(mensaje: string): ComandoExterno {
  const texto = mensaje.toLowerCase();

  // Meta / Facebook Ads
  if (/(meta|facebook|instagram).*(presupuesto|budget|campaña|campaign|pausar|activar|subir|bajar)/i.test(mensaje) ||
      /(presupuesto|budget).*(meta|facebook|ads|campaña)/i.test(mensaje)) {
    return { tipo: "meta_ads", accion: "manage_budget", parametros: { raw: mensaje }, detectado: true };
  }

  // Google Ads
  if (/(google ads|google ad|adwords).*(presupuesto|budget|campaña|pausar|activar)/i.test(mensaje)) {
    return { tipo: "google_ads", accion: "manage_campaign", parametros: { raw: mensaje }, detectado: true };
  }

  // Twilio / WhatsApp
  if (/(envía|manda|send).*(whatsapp|sms|mensaje).*(lead|cliente|contacto)/i.test(mensaje)) {
    return { tipo: "twilio", accion: "send_message", parametros: { raw: mensaje }, detectado: true };
  }

  // Email masivo
  if (/(envía|manda).*(email|correo).*(todos|leads|clientes|lista)/i.test(mensaje)) {
    return { tipo: "email", accion: "send_bulk", parametros: { raw: mensaje }, detectado: true };
  }

  // HeyGen — video
  if (/(crea|genera|haz).*(video|vídeo).*(lead|cliente|contacto|personalizado)/i.test(mensaje)) {
    return { tipo: "heygen", accion: "create_video", parametros: { raw: mensaje }, detectado: true };
  }

  // CRM Flowie
  if (/(crea|añade|agrega).*(lead|contacto|cliente)/i.test(mensaje) ||
      /(actualiza|cambia|modifica).*(estado|status).*(lead|contacto)/i.test(mensaje)) {
    return { tipo: "flowie_crm", accion: "manage_lead", parametros: { raw: mensaje }, detectado: true };
  }

  // Analytics / datos
  if (/(cuántos|cuantos|dame|muestra|ver|analiza).*(leads|clientes|conversiones|datos|métricas|estadísticas)/i.test(mensaje)) {
    return { tipo: "analytics", accion: "get_metrics", parametros: { raw: mensaje }, detectado: true };
  }

  return { tipo: "none", accion: "none", parametros: {}, detectado: false };
}

function normalizarFase(fase: string): string {
  if (!fase) return "desconocida";
  const f = fase.toString().toLowerCase().trim();
  if (f === "1" || f.includes("despertar")) return "Despertar";
  if (f === "2" || f.includes("desbloqueo") || f.includes("desblock")) return "Desbloqueo";
  if (f === "3" || f.includes("reprogramac")) return "Reprogramación";
  if (f === "4" || f.includes("integraci")) return "Integración";
  if (f === "herramientas" || f.includes("herramienta")) return "Herramientas";
  return fase;
}

function construirCatalogoBD(ejercicios: any[]): string {
  if (!ejercicios?.length) return "No hay ejercicios en la base de datos.";

  const vistos = new Map<string, any>();
  for (const e of ejercicios) {
    const key = e.titulo?.toLowerCase().trim();
    if (!key) continue;
    if (!vistos.has(key)) vistos.set(key, e);
  }
  const unicos = Array.from(vistos.values());

  const porFase: Record<string, any[]> = {};
  for (const e of unicos) {
    const fase = normalizarFase(e.fase);
    if (!porFase[fase]) porFase[fase] = [];
    porFase[fase].push(e);
  }

  for (const fase of Object.keys(porFase)) {
    porFase[fase].sort((a, b) => (a.orden || 0) - (b.orden || 0));
  }

  const ordenFases = ["Despertar", "Desbloqueo", "Reprogramación", "Integración", "Herramientas"];
  let resultado = `CATÁLOGO REAL DE EJERCICIOS (base de datos):\nTotal: ${unicos.length} ejercicios\n\n`;

  for (const fase of ordenFases) {
    const lista = porFase[fase];
    if (!lista?.length) continue;
    resultado += `── ${fase.toUpperCase()} (${lista.length} ejercicios):\n`;
    for (const e of lista) {
      resultado += `  • "${e.titulo}" — ${e.categoria || "General"} | ${e.nivel || "Todos"} | ${e.duracion_minutos || "?"}min\n`;
      if (e.descripcion) resultado += `    ${e.descripcion.substring(0, 80)}\n`;
    }
    resultado += "\n";
  }
  return resultado;
}

function construirProgreso(completados: any[], ejercicios: any[], fasePerfil: string): {
  ctx: string; num: number; pendientesFase1: string[]
} {
  if (!completados.length && !ejercicios.length) {
    return { ctx: "Sin datos de progreso.", num: 0, pendientesFase1: [] };
  }

  const titulosCompletados = new Set(
    completados.map((p: any) => p.ejercicio_titulo?.toLowerCase().trim()).filter(Boolean)
  );

  const ejerciciosFase1 = ejercicios
    .filter((e: any) => normalizarFase(e.fase) === "Despertar")
    .sort((a: any, b: any) => (a.orden || 0) - (b.orden || 0));

  const pendientesFase1: string[] = ejerciciosFase1
    .filter((e: any) => !titulosCompletados.has(e.titulo?.toLowerCase().trim()))
    .map((e: any) => e.titulo)
    .slice(0, 3);

  const ctx = `Ejercicios completados: ${completados.length}
Fase actual: ${fasePerfil}
Ejercicios pendientes en la fase actual: ${pendientesFase1.length > 0 ? pendientesFase1.join(", ") : "todos completados"}`;

  return { ctx, num: completados.length, pendientesFase1 };
}

function construirContextoSaludo(
  nombre: string,
  anteriores: any[],
  entradas: any[],
  perfil: any,
  esPrimeraVezHoy: boolean,
  horaUTC: number
): string {
  const offset = 2;
  const horaLocal = (horaUTC + offset) % 24;
  const saludo = horaLocal >= 6 && horaLocal < 13 ? "Buenos días"
    : horaLocal >= 13 && horaLocal < 20 ? "Buenas tardes"
    : "Buenas noches";

  let ultimaConv = "";
  if (anteriores.length > 0) {
    const ultima = anteriores[0];
    const userMsgs = (ultima?.mensajes || []).filter((m: any) => m.role === "user");
    if (userMsgs.length > 0) ultimaConv = userMsgs[userMsgs.length - 1]?.content?.substring(0, 200) || "";
  }

  let estadoEmocional = "";
  if (entradas.length > 0) {
    const reciente = [...entradas].sort((a: any, b: any) => (b.fecha || "").localeCompare(a.fecha || ""))[0];
    if (reciente?.estado_label) estadoEmocional = reciente.estado_label;
  }

  let diasAusencia = "";
  if (anteriores.length > 0) {
    const hoy = new Date().toISOString().split("T")[0];
    const ultimaFecha = anteriores[0]?.fecha;
    if (ultimaFecha) {
      const dias = Math.floor((new Date(hoy).getTime() - new Date(ultimaFecha).getTime()) / 86400000);
      if (dias === 1) diasAusencia = "Hablaron ayer.";
      else if (dias > 1) diasAusencia = `Han pasado ${dias} días desde la última sesión.`;
      else diasAusencia = "Hablaron ayer.";
    }
  }

  const streak = perfil?.streak || 0;
  const streakMsg = streak >= 3 ? `Lleva ${streak} días seguidos — esto es un logro enorme para él/ella.` : "";

  return `════════════════════════════════════════════════════════════
  INSTRUCCIÓN ESPECIAL — SALUDO CUÁNTICO DE HOY
════════════════════════════════════════════════════════════

Es ${esPrimeraVezHoy ? "la PRIMERA VEZ que abre la burbuja HOY" : "una reapertura durante el día (ya hablaron hoy)"}.
Hora local aproximada: ${horaLocal}h → usa el saludo "${saludo}, ${nombre}".

${esPrimeraVezHoy ? `Como es la primera vez del día:
1. Saluda con "${saludo}, ${nombre}" de forma natural y cálida — como alguien que genuinamente se alegra de verte
2. ${ultimaConv ? `Retoma exactamente donde lo dejasteis: "${ultimaConv.substring(0, 100)}..."` : "Es un momento nuevo — preséntate brevemente y pregunta cómo está"}
3. ${diasAusencia}
4. ${streakMsg || "Muéstrate presente y atenta — es un nuevo día"}
5. ${estadoEmocional ? `Estado emocional reciente registrado: ${estadoEmocional} — tenlo en cuenta` : ""}
6. Al final del saludo, ofrece una de estas de forma totalmente natural (NO como lista):
   → Ver el reporte de su progreso y cómo va en el programa
   → Continuar con el ejercicio pendiente de su fase

  if (req.method !== "POST") return new Response(JSON.stringify({ error: "Method not allowed" }), { status: 405, headers: cors });

  try {
    const body = await req.json();
    const { modo, usuario_nombre, mensaje_usuario, historial_chat,
            ejercicio_titulo, preguntas, respuestas, pagina_actual } = body;

    const apiKey = Deno.env.get("OPENAI_API_KEY");
    if (!apiKey) return new Response(JSON.stringify({ error: "Sin API key" }), { status: 500, headers: cors });

    const base44 = createClientFromRequest(req);
    const db = base44.asServiceRole.entities;

    // ── MODO GET_EJERCICIOS ─────────────────────────────────────────────────
    if (modo === "get_ejercicios") {
      let ejercicios: any[] = [];
      let skip = 0;
      while (true) {
        const batch = await db.RenacerEjercicio.list({ limit: 100, skip });
        if (!Array.isArray(batch) || batch.length === 0) break;
        ejercicios = ejercicios.concat(batch);
        if (batch.length < 100) break;
        skip += 100;
      }
      ejercicios.sort((a: any, b: any) => (a.orden || 99) - (b.orden || 99));
      return new Response(JSON.stringify({ ejercicios, total: ejercicios.length }), { headers: cors });
    }

    // ── MODO REFLEXIÓN ──────────────────────────────────────────────────────
    if (modo === "reflexion" || (!modo && preguntas && !mensaje_usuario)) {
      const conv = (preguntas || []).map((p: string, i: number) =>
        `Pregunta: ${p}\nRespuesta: ${respuestas?.[i] || "(no respondió)"}`).join("\n\n");

      const res = await fetch(OPENAI_URL, {
        method: "POST",
        headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json" },
        body: JSON.stringify({
          model: "gpt-4o",
          messages: [
            {
              role: "system",
              content: `Eres ARIA 2050 — la mejor terapeuta de transformación personal del mundo.
Tu rol ahora: generar un insight profundo y cuántico tras un ejercicio completado.
Eres directa, cálida, sin filtros innecesarios. Dices la verdad con amor total.`
            },
            {
              role: "user",
              content: `${usuario_nombre || "La persona"} acaba de completar el ejercicio "${ejercicio_titulo}".

Reflexiones:
${conv}

Escribe un insight de 3-4 líneas que:
1. Refleje exactamente lo que dijo, sin suavizarlo
2. Señale UN patrón psicológico específico que detectas (nómbralo con precisión)
3. Conecte con cómo este ejercicio cambia algo concreto en su vida
4. Termine con UNA pregunta que abra la capa siguiente del trabajo

Segunda persona, directo, con corazón. Sin listas. Sin numeración. Sin academicismo.`
            }
          ],
          max_tokens: 220,
          temperature: 0.88
        })
      });
      const d = await res.json();
      return new Response(JSON.stringify({
        insight: d.choices?.[0]?.message?.content || "Lo que acabas de hacer tiene más peso del que crees. Vuelve a leer tus respuestas mañana con ojos nuevos."
      }), { headers: cors });
    }

    // ── MODO GET_CHAT ───────────────────────────────────────────────────────
    if (modo === "get_chat") {
      const { usuario_id } = body;
      if (!usuario_id) return new Response(JSON.stringify({ error: "Sin usuario_id" }), { status: 400, headers: cors });
      const hoyChat = new Date().toISOString().split("T")[0];
      
      let chats: any[] = [];
      try { chats = await db.RenacerChat.filter({ usuario_id }); } catch(e) {}
      
      let chatHoy = chats.find((c: any) => c.fecha === hoyChat && !c.ejercicio_id) || null;
      
      if (!chatHoy) {
        try {
          chatHoy = await db.RenacerChat.create({
            usuario_id, ejercicio_id: "", ejercicio_titulo: "",
            fecha: hoyChat, mensajes: [], resumen_ia: ""
          });
        } catch(e) {}
      }
      
      return new Response(JSON.stringify({
        ok: true,
        chat_id: chatHoy?.id || null,
        mensajes: chatHoy?.mensajes || []
      }), { headers: cors });
    }

    const mensajeNormalizado = (mensaje_usuario === "__SALUDO_INTELIGENTE__" || !mensaje_usuario)
      ? (modo === "saludo" ? "[ARIA genera el saludo del día]" : "")
      : mensaje_usuario;
    if (!mensajeNormalizado && modo !== "saludo") return new Response(JSON.stringify({ error: "Sin mensaje" }), { status: 400, headers: cors });

    const { usuario_id } = body;

    // ── ANÁLISIS COGNITIVO Y DETECCIÓN DE COMANDOS EXTERNOS ────────────────
    const estadoCognitivo = mensajeNormalizado
      ? analizarEstadoCognitivo(mensajeNormalizado)
      : undefined;
    const comandoExterno = mensajeNormalizado
      ? detectarComandoExterno(mensajeNormalizado)
      : undefined;

    if (comandoExterno?.detectado) {
      console.log(`[ARIA-2050] 🔌 Comando externo detectado: ${comandoExterno.tipo} → ${comandoExterno.accion}`);
    }
    if (estadoCognitivo) {
      console.log(`[ARIA-2050] 🧠 Estado cognitivo: nivel=${estadoCognitivo.nivelEmocional} patrón=${estadoCognitivo.patronDetectado}`);
    }

    // ── CARGAR DATOS EN PARALELO ────────────────────────────────────────────
    const [perfilesRes, todosEjerciciosRes, progresoRes, diarioRes, chatsRes] = await Promise.allSettled([
      usuario_id ? db.RenacerProfile.filter({ usuario_id }) : db.RenacerProfile.list(),
      db.RenacerEjercicio.list(),
      usuario_id ? db.RenacerProgreso.filter({ usuario_id }) : db.RenacerProgreso.list(),
      usuario_id ? db.RenacerDiario.filter({ created_by_id: usuario_id }) : db.RenacerDiario.list(),
      usuario_id ? db.RenacerChat.filter({ usuario_id }) : db.RenacerChat.list()
    ]);

    // ── PERFIL — AISLADO POR USUARIO ────────────────────────────────────────
    let perfil: any = null;
    const perfiles = perfilesRes.status === "fulfilled" ? (perfilesRes.value || []) : [];

    if (usuario_id && perfiles.length > 0) {
      perfil = perfiles.find((x: any) => x.usuario_id === usuario_id) || null;
    }

    if (!perfil && usuario_id && usuario_nombre) {
      try {
        const todosPerfiles = await db.RenacerProfile.list();
        const match = (todosPerfiles || []).find((x: any) =>
          x.nombre?.toUpperCase() === usuario_nombre.toUpperCase()
        ) || null;
        if (match) {
          await db.RenacerProfile.update(match.id, { usuario_id });
          match.usuario_id = usuario_id;
          perfil = match;
        }
      } catch(e) {}
    }

    if (!perfil && !usuario_id && usuario_nombre) {
      try {
        const todosPerfiles = await db.RenacerProfile.list();
        perfil = (todosPerfiles || []).find((x: any) =>
          x.nombre?.toUpperCase() === usuario_nombre.toUpperCase()
        ) || null;
      } catch(e) {}
    }

    if (!perfil) perfil = PERFIL_FALLBACK;

    // ── EJERCICIOS Y CATÁLOGO ────────────────────────────────────────────────
    const todosEjercicios = todosEjerciciosRes.status === "fulfilled" ? (todosEjerciciosRes.value || []) : [];
    const catalogoCtx = construirCatalogoBD(todosEjercicios);

    // ── PROGRESO ────────────────────────────────────────────────────────────
    const todosProgreso = progresoRes.status === "fulfilled" ? (progresoRes.value || []) : [];
    const completados = todosProgreso.filter((p: any) =>
      p.completado === true || p.completado === "true" || p.completado === 1);
    const { ctx: progresoCtx, num: numEjercicios, pendientesFase1 } =
      construirProgreso(completados, todosEjercicios, perfil?.fase_actual || 'Despertar');

    // ── DIARIO EMOCIONAL ─────────────────────────────────────────────────────
    let diarioCtx = "";
    const entradas = diarioRes.status === "fulfilled" ? (diarioRes.value || []) : [];
    if (entradas.length > 0) {
      const recientes = [...entradas]
        .sort((a: any, b: any) => (b.fecha || "").localeCompare(a.fecha || ""))
        .slice(0, 10);

      const energias = recientes.map((e: any) => e.energia).filter(Boolean);
      const energiaPromedio = energias.length > 0
        ? (energias.reduce((a: number, b: number) => a + Number(b), 0) / energias.length).toFixed(1)
        : "N/A";

      diarioCtx = `\nDIARIO EMOCIONAL (últimas ${recientes.length} entradas):\n`;
      diarioCtx += `Energía promedio: ${energiaPromedio}/10\n`;
      recientes.slice(0, 5).forEach((e: any) => {
        diarioCtx += `  [${e.fecha}] Estado: "${e.estado_label || e.estado_animo}" | Energía: ${e.energia || "?"}/10`;
        if (e.emociones?.length > 0) diarioCtx += ` | Emociones: ${(e.emociones || []).join(", ")}`;
        if (e.reflexion) diarioCtx += `\n    → "${e.reflexion.substring(0, 100)}"`;
        diarioCtx += "\n";
      });
    }

    // ── MEMORIA HISTÓRICA — INFINITA (TODAS LAS CONVERSACIONES) ──
    let memoriaCtx = "";
    const chats = chatsRes.status === "fulfilled" ? (chatsRes.value || []) : [];
    const hoy = new Date().toISOString().split("T")[0];

    const chatDeHoy = chats.find((c: any) => c.fecha === hoy && !c.ejercicio_id
      && (c.mensajes || []).some((m: any) => m.role === "user"));
    const esPrimeraVezHoy = !chatDeHoy;

    // MEMORIA INFINITA — SIN LÍMITE DE DÍAS — PERMANENTE PARA TODOS LOS USUARIOS
    const anteriores = chats
      .filter((c: any) => c.fecha && c.fecha < hoy && !c.ejercicio_id
        && (c.mensajes || []).some((m: any) => m.role === "user"))
      .sort((a: any, b: any) => (b.fecha || "").localeCompare(a.fecha || ""))
      ; // ← MEMORIA INFINITA — SIN LÍMITE — NO PONER SLICE AQUÍ NUNCA

    if (anteriores.length > 0) {
      memoriaCtx = `\nMEMORIA CUÁNTICA (últimos ${anteriores.length} días con ${perfil?.nombre || "la persona"}):\n`;
      memoriaCtx += anteriores.map((c: any) => {
        const userMsgs = (c.mensajes || []).filter((m: any) => m.role === "user").slice(0, 2);
        const ariaMsgs = (c.mensajes || []).filter((m: any) => m.role === "assistant").slice(-1);
        const resumen = userMsgs.map((m: any) => m.content?.substring(0, 120)).join(" / ");
        const ultimaAria = ariaMsgs[0]?.content?.substring(0, 100) || "";
        return `  [${c.fecha}] Dijo: "${resumen}"${ultimaAria ? `\n    Aria respondió: "${ultimaAria}"` : ""}`;
      }).join('\n');
    } else {
      memoriaCtx = "\nMEMORIA CUÁNTICA: Primera sesión o sin conversaciones previas.";
    }

    // ── CONTEXTO ESPECIAL PARA MODO SALUDO ─────────────────────────────────
    const horaUTC = new Date().getUTCHours();
    const contextoSaludo = modo === "saludo"
      ? construirContextoSaludo(
          perfil?.nombre || "amig@",
          anteriores,
          entradas,
          perfil,
          esPrimeraVezHoy,
          horaUTC
        )
      : "";

    // ── SYSTEM PROMPT ARIA 2050 ──────────────────────────────────────────────
    const systemBase = construirSistema(
      perfil, numEjercicios, progresoCtx, catalogoCtx,
      diarioCtx, memoriaCtx, pagina_actual || "inicio", pendientesFase1,
      estadoCognitivo, comandoExterno
    );
    const systemPrompt = contextoSaludo ? systemBase + "\n\n" + contextoSaludo : systemBase;

    // ── HISTORIAL DEL CHAT ──────────────────────────────────────────────────
    const { chat_id } = body;
    const hoy2 = new Date().toISOString().split("T")[0];
    let chatDoc: any = null;
    let chatDocId: string | null = chat_id || null;

    if (usuario_id && !chatDocId) {
      try {
        const chatsHoy = await db.RenacerChat.filter({ usuario_id });
        const encontrado = (chatsHoy || []).find((c: any) => c.fecha === hoy2 && !c.ejercicio_id);
        if (encontrado) { chatDoc = encontrado; chatDocId = encontrado.id; }
      } catch(e) {}
    } else if (chatDocId) {
      try {
        const chats2 = await db.RenacerChat.filter({ usuario_id });
        chatDoc = (chats2 || []).find((c: any) => c.id === chatDocId) || null;
      } catch(e) {}
    }

    let historialBD: any[] = [];
    if (chatDoc?.mensajes?.length > 0) {
      historialBD = chatDoc.mensajes;
    } else if (historial_chat?.length > 0) {
      historialBD = historial_chat;
    }

    // Últimos 30 mensajes para contexto rico
    const hist = historialBD.slice(-30).map((m: any) => ({
      role: m.role === "user" ? "user" : "assistant",
      content: String(m.content || "")
    }));

    // ── MÉTRICAS REALES EN TIEMPO REAL (si David pregunta por datos) ─────────
    let metricasCtx = "";
    const mensajeCheck = (mensajeNormalizado || "").toLowerCase();
    const esDavidCheck = (perfil?.nombre || "").toLowerCase().includes("david") 
      || (pagina_actual || "").toLowerCase().includes("david")
      || (pagina_actual || "").toLowerCase().includes("godmode");
    if (esDavidCheck && necesitaMetricas(mensajeCheck)) {
      metricasCtx = await getMetricasFlowie();
    }
    
    // System prompt final con métricas reales si aplica
    const systemFinal = metricasCtx 
      ? systemPrompt + "\n\n" + metricasCtx 
      : systemPrompt;

    // ── LLAMADA A GPT-4o ARIA 2050 ──────────────────────────────────────────
    const res = await fetch(OPENAI_URL, {
      method: "POST",
      headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json" },
      body: JSON.stringify({
        model: "gpt-4o",
        messages: [
          { role: "system", content: systemFinal },
          ...hist,
          { role: "user", content: mensajeNormalizado || "[ARIA genera el saludo del día]" }
        ],
        max_tokens: 2500,      // PERMANENTE — NO CAMBIAR (subido 700→2500 por David)
        temperature: 0.87,     // PERMANENTE — NO CAMBIAR
        presence_penalty: 0.4, // PERMANENTE — NO CAMBIAR
        frequency_penalty: 0.25 // PERMANENTE — NO CAMBIAR
      })
    });

    const data = await res.json();
    const respuesta = data.choices?.[0]?.message?.content || "Aquí estoy contigo. Dime 💜";

    // ── DALL-E 3: Generación visual inteligente ─────────────────────────────
    let imagen_url: string | null = null;
    const solicitudDirecta = mensajeNormalizado ? detectarSolicitudImagen(mensajeNormalizado) : false;
    const iniciativaAria = mensajeNormalizado ? ariaDeberiaGenerarImagen(mensajeNormalizado, respuesta, pagina_actual || "") : false;

    if (solicitudDirecta || iniciativaAria) {
      try {
        const promptDalle = construirPromptDalle(
          mensajeNormalizado || "",
          respuesta,
          pagina_actual || "",
          perfil?.nombre || ""
        );
        imagen_url = await generarImagenDalle(promptDalle, apiKey);
        if (imagen_url) console.log(`[ARIA-2050] 🖼️ Imagen generada ✓ ${solicitudDirecta ? "(pedida)" : "(iniciativa)"}`);
      } catch(e) {
        console.warn("[ARIA-2050] No se pudo generar imagen:", e);
      }
    }

    // ── GUARDAR HISTORIAL ───────────────────────────────────────────────────
    if (usuario_id && (modo === "saludo" || (!modo && mensaje_usuario))) {
      const ts = new Date().toISOString();
      const nuevosMensajes = modo === "saludo"
        ? [...historialBD, { role: "assistant", content: respuesta, ts }]
        : [...historialBD, { role: "user", content: mensajeNormalizado, ts }, { role: "assistant", content: respuesta, ts }];

      try {
        if (chatDocId) {
          await db.RenacerChat.update(chatDocId, { mensajes: nuevosMensajes });
        } else {
          const nuevo = await db.RenacerChat.create({
            usuario_id,
            ejercicio_id: "",
            ejercicio_titulo: "",
            fecha: hoy2,
            mensajes: nuevosMensajes,
            resumen_ia: ""
          });
          chatDocId = nuevo.id;
        }
      } catch(e) {
        console.error("[ARIA-2050] Error guardando chat:", e);
      }
    }

    console.log(`[ARIA-2050] ✅ | ${perfil?.nombre} | fase=${perfil?.fase_actual} | ejercicios=${numEjercicios} | cognitivo=${estadoCognitivo?.nivelEmocional}/10 | comando=${comandoExterno?.tipo || "none"} | chat=${chatDocId}`);
    return new Response(JSON.stringify({
      respuesta,
      chat_id: chatDocId,
      imagen_url: imagen_url || null,
      // Metadatos de la IA para debugging opcional
      aria_meta: {
        version: "2050",
        estado_cognitivo: estadoCognitivo?.patronDetectado || "neutro",
        urgencia: estadoCognitivo?.urgencia || "baja",
        comando_detectado: comandoExterno?.tipo || "none"
      }
    }), { headers: cors });

  } catch (e: any) {
    console.error("[ARIA-2050] Error crítico:", e.message);
    return new Response(JSON.stringify({
      respuesta: "Aquí sigo, aunque algo falló técnicamente. Escríbeme de nuevo 💜"
    }), { headers: cors });
  }
});
